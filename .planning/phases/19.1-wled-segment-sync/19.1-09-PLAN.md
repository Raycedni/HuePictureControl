**Step 1: Create `Frontend/e2e/wled-segments.spec.ts`**

```typescript
import { test, expect, type Page } from '@playwright/test'

const BACKEND = process.env.BACKEND_URL ?? 'http://localhost:8000'
const APP = process.env.APP_URL ?? 'http://localhost:8091'

async function ensureWledDevice(page: Page): Promise<{ id: string; ledCount: number }> {
  const resp = await page.request.get(`${BACKEND}/api/wled/devices`)
  expect(resp.ok()).toBeTruthy()
  const body = await resp.json()
  if (body.devices && body.devices.length > 0) {
    const d = body.devices[0]
    return { id: d.id, ledCount: d.led_count }
  }
  test.skip(true, 'No WLED device registered; run manual UAT to verify hardware-side.')
  return { id: '', ledCount: 0 }
}

test.describe('Phase 19.1 WLED segments', () => {
  test('segments render from refresh response (stubbed)', async ({ page }) => {
    const device = await ensureWledDevice(page)

    await page.route(`**/api/wled/devices/${device.id}/segments/refresh`, (route) =>
      route.fulfill({
        status: 200,
        contentType: 'application/json',
        body: JSON.stringify({
          segments: [
            { seg_index: 0, start_led: 0, stop_led: 99, name: 'Sofa', refreshed_at: new Date().toISOString() },
            { seg_index: 1, start_led: 100, stop_led: 199, name: null, refreshed_at: new Date().toISOString() },
            { seg_index: 2, start_led: 200, stop_led: 299, name: 'TV', refreshed_at: new Date().toISOString() },
          ],
          dropped_assignments: 0,
        }),
      }),
    )

    // Also stub the initial GET so the page load doesn't depend on whatever cache state the backend has.
    await page.route(`**/api/wled/devices/${device.id}/segments`, (route) =>
      route.fulfill({
        status: 200,
        contentType: 'application/json',
        body: JSON.stringify({ segments: [] }),
      }),
    )

    await page.goto(`${APP}/settings/wled`)  // adjust path per actual routing
    await page.getByTestId(`wled-refresh-button-${device.id}`).click()

    await expect(page.getByTestId(`wled-seg-${device.id}-0`)).toBeVisible()
    await expect(page.getByTestId(`wled-seg-${device.id}-1`)).toBeVisible()
    await expect(page.getByTestId(`wled-seg-${device.id}-2`)).toBeVisible()
  })

  test('fit-to-width per device', async ({ page }) => {
    const device = await ensureWledDevice(page)

    await page.route(`**/api/wled/devices/${device.id}/segments`, (route) =>
      route.fulfill({
        status: 200,
        contentType: 'application/json',
        body: JSON.stringify({
          segments: [
            { seg_index: 0, start_led: 0, stop_led: device.ledCount - 1, name: 'Full', refreshed_at: new Date().toISOString() },
          ],
        }),
      }),
    )

    await page.goto(`${APP}/settings/wled`)

    const stripContainer = page.getByTestId('wled-strip-painter').first()
    await expect(stripContainer).toBeVisible()
    const box = await stripContainer.boundingBox()
    expect(box).not.toBeNull()
    expect(box!.width).toBeGreaterThan(100)  // Sanity: the strip occupies meaningful width

    const zone = page.getByTestId(`wled-seg-${device.id}-0`)
    await expect(zone).toBeVisible()
  })

  test('drag payload uses wledDeviceId + seg_index per D-13', async ({ page }) => {
    const device = await ensureWledDevice(page)

    // Stub segments via the GET — LightPanel reads from list, not from refresh
    await page.route(`**/api/wled/devices/${device.id}/segments`, (route) =>
      route.fulfill({
        status: 200,
        contentType: 'application/json',
        body: JSON.stringify({
          segments: [{ seg_index: 0, start_led: 0, stop_led: 9, name: 'Sofa', refreshed_at: new Date().toISOString() }],
        }),
      }),
    )

    await page.goto(`${APP}/editor`)

    // Capture the drag payload via a small evaluator that hooks dataTransfer.setData.
    const captured = await page.evaluate((deviceId) => {
      return new Promise<Record<string, string>>((resolve) => {
        const data: Record<string, string> = {}
        const sel = `[data-testid="lightpanel-wled-seg-${deviceId}-0"]`
        const row = document.querySelector(sel) as HTMLElement | null
        if (!row) {
          resolve({})
          return
        }
        const evt = new Event('dragstart', { bubbles: true })
        // Build a fake dataTransfer
        ;(evt as DragEvent & { dataTransfer: DataTransfer }).dataTransfer = {
          setData: (key: string, value: string) => { data[key] = value },
          effectAllowed: 'none',
        } as unknown as DataTransfer
        row.dispatchEvent(evt)
        setTimeout(() => resolve(data), 50)
      })
    }, device.id)

    expect(captured).toHaveProperty('wledDeviceId', device.id)
    expect(captured).toHaveProperty('seg_index', '0')
    expect(captured).toHaveProperty('wledSegName', 'Sofa')
  })
})
```

Note the third spec uses a manual `dispatchEvent` because Playwright's native drag-and-drop is awkward to verify payload-only. If the existing `wled-paint.spec.ts` had a different working drag-payload-capture pattern, copy that one instead.
