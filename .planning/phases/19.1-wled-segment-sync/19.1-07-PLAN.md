```typescript
function staleBadge(segs: WledSegment[] | undefined): string | null {
  if (!segs || segs.length === 0) return null
  const ts = segs[0]?.refreshed_at
  if (!ts) return null
  const ageMs = Date.now() - new Date(ts).getTime()
  if (Number.isNaN(ageMs)) return null
  if (ageMs < 60_000) {
    return `synced ${Math.max(1, Math.round(ageMs / 1000))}s ago`
  }
  return `stale (${Math.round(ageMs / 60_000)}m ago)`
}
```

```typescript
import { useEffect, useState } from 'react'
import { listSegments, WledApiError } from '@/api/wled'
import type { WledSegment } from '@/utils/wled-segment'
import { segmentName } from '@/utils/wled-segment'

interface WledChannelSidebarProps {
  selectedSeg: { device_id: string; seg_index: number } | null
  onClear?: () => void
}

export function WledChannelSidebar({ selectedSeg, onClear }: WledChannelSidebarProps) {
  const [seg, setSeg] = useState<WledSegment | null>(null)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    if (!selectedSeg) {
      setSeg(null)
      setError(null)
      return
    }
    let alive = true
    listSegments(selectedSeg.device_id)
      .then((resp) => {
        if (!alive) return
        const found = resp.segments.find((s) => s.seg_index === selectedSeg.seg_index)
        setSeg(found ?? null)
        setError(found ? null : 'Segment not found in cache')
      })
      .catch((err) => {
        if (!alive) return
        setError(err instanceof WledApiError ? `HTTP ${err.status}` : 'Failed to load segment')
      })
    return () => { alive = false }
  }, [selectedSeg?.device_id, selectedSeg?.seg_index])

  if (!selectedSeg) {
    return (
      <div
        className="rounded-lg bg-white/[0.03] border border-white/[0.06] p-3 text-xs text-muted-foreground"
        data-testid="wled-channel-sidebar-empty"
      >
        <h3 className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider mb-2">
          Selected segment
        </h3>
        Select a zone on the strip to view it.
      </div>
    )
  }

  if (error) {
    return (
      <div
        className="rounded-lg bg-white/[0.03] border border-white/[0.06] p-3 text-xs text-destructive"
        data-testid="wled-channel-sidebar-error"
      >
        {error}
      </div>
    )
  }

  if (!seg) {
    return (
      <div
        className="rounded-lg bg-white/[0.03] border border-white/[0.06] p-3 text-xs text-muted-foreground"
        data-testid="wled-channel-sidebar-loading"
      >
        Loading…
      </div>
    )
  }

  const length = seg.stop_led - seg.start_led + 1
  return (
    <div
      className="rounded-lg bg-white/[0.03] border border-white/[0.06] p-3 text-xs"
      data-testid="wled-channel-sidebar"
    >
      <h3 className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider mb-2">
        Selected segment
      </h3>
      <dl className="space-y-1">
        <div className="flex justify-between">
          <dt className="text-muted-foreground">Name</dt>
          <dd data-testid="sidebar-seg-name">{segmentName(seg)}</dd>
        </div>
        <div className="flex justify-between">
          <dt className="text-muted-foreground">Index</dt>
          <dd data-testid="sidebar-seg-index">{seg.seg_index}</dd>
        </div>
        <div className="flex justify-between">
          <dt className="text-muted-foreground">Range</dt>
          <dd data-testid="sidebar-seg-range">{seg.start_led}–{seg.stop_led}</dd>
        </div>
        <div className="flex justify-between">
          <dt className="text-muted-foreground">Length</dt>
          <dd data-testid="sidebar-seg-length">{length} LEDs</dd>
        </div>
      </dl>
      {onClear && (
        <button
          type="button"
          onClick={onClear}
          className="mt-3 text-xs text-muted-foreground hover:text-foreground"
          data-testid="wled-channel-sidebar-clear"
        >
          Clear selection
        </button>
      )}
    </div>
  )
}
```

Note: a `<button>` for "Clear selection" is acceptable per D-07 because it only calls a UI callback (selection clear); it does NOT mutate channel/segment data.
